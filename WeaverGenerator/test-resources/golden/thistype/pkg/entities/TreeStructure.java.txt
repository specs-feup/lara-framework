package thistype.pkg.entities;

import java.util.List;
import thistype.pkg.abstracts.joinpoints.AJoinPoint;
import java.util.Map;

/**
 * 
 * 
 * @author Lara C.
 */
public class TreeStructure {

    private Integer depth;
    private List<List<AJoinPoint<?>>> levels;
    private Map<String, AJoinPoint<?>> nodeIndex;
    private AJoinPoint<?> root;

    /**
     * 
     */
    public TreeStructure(){}
    /**
     * 
     */
    public TreeStructure(Integer depth, List<List<AJoinPoint<?>>> levels, Map<String, AJoinPoint<?>> nodeIndex, AJoinPoint<?> root){
        this.setDepth(depth);
        this.setLevels(levels);
        this.setNodeIndex(nodeIndex);
        this.setRoot(root);
    }
    /**
     * Get value on attribute depth
     * @return the attribute's value
     */
    public Integer getDepth() {
        return this.depth;
    }

    /**
     * Set value on attribute depth
     * @param depth
     */
    public void setDepth(Integer depth) {
        this.depth = depth;
    }

    /**
     * Get value on attribute levels
     * @return the attribute's value
     */
    public List<List<AJoinPoint<?>>> getLevels() {
        return this.levels;
    }

    /**
     * Set value on attribute levels
     * @param levels
     */
    public void setLevels(List<List<AJoinPoint<?>>> levels) {
        this.levels = levels;
    }

    /**
     * Get value on attribute nodeIndex
     * @return the attribute's value
     */
    public Map<String, AJoinPoint<?>> getNodeIndex() {
        return this.nodeIndex;
    }

    /**
     * Set value on attribute nodeIndex
     * @param nodeIndex
     */
    public void setNodeIndex(Map<String, AJoinPoint<?>> nodeIndex) {
        this.nodeIndex = nodeIndex;
    }

    /**
     * Get value on attribute root
     * @return the attribute's value
     */
    public AJoinPoint<?> getRoot() {
        return this.root;
    }

    /**
     * Set value on attribute root
     * @param root
     */
    public void setRoot(AJoinPoint<?> root) {
        this.root = root;
    }

    /**
     * 
     */
    @Override
    public String toString() {
        String json = "{\n";
        json += " depth: "+getDepth() + ",\n";
        json += " levels: "+getLevels() + ",\n";
        json += " nodeIndex: "+getNodeIndex() + ",\n";
        json += " root: "+getRoot() + ",\n";
        json+="}";
        return json;
    }
}
