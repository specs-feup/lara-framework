package thistype.pkg.enums;

import org.lara.interpreter.weaver.interf.NamedEnum;
import pt.up.fe.specs.util.lazy.Lazy;
import pt.up.fe.specs.util.enums.EnumHelperWithValue;

/**
 * 
 * 
 * @author Lara C.
 */
public enum NodeKind  implements NamedEnum{
    DECLARATION("declaration"),
    EXPRESSION("expression"),
    OTHER("other"),
    STATEMENT("statement");
    private String name;
    private static final Lazy<EnumHelperWithValue<NodeKind>> ENUM_HELPER = EnumHelperWithValue.newLazyHelperWithValue(NodeKind.class);

    /**
     * 
     */
    private NodeKind(String name){
        this.name = name;
    }
    /**
     * 
     */
    public String getName() {
        return this.name;
    }

    /**
     * 
     */
    public String toString() {
        return getName();
    }

    /**
     * 
     */
    public static EnumHelperWithValue<NodeKind> getHelper() {
        return ENUM_HELPER.get();
    }
}
